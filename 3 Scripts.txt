local function safeDestroy(object)
    if object then
        pcall(function()
            object:Destroy()
        end)
    end
end

--// Clipboard helper (works across executors)
local function copyToClipboard(text)
    local ok = false
    pcall(function()
        if setclipboard then
            setclipboard(text); ok = true
        elseif syn and syn.setclipboard then
            syn.setclipboard(text); ok = true
        elseif Clipboard and Clipboard.set then
            Clipboard.set(text); ok = true
        end
    end)
    return ok
end

local Dragging = false
local DragStart
local StartPosition
local DragInput

MainFrame.InputBegan:Connect(function(input)
    if input.UserInputType == Enum.UserInputType.MouseButton1
        or input.UserInputType == Enum.UserInputType.Touch then

        Dragging = true
        DragStart = input.Position
        StartPosition = MainFrame.Position
    end
end)

MainFrame.InputChanged:Connect(function(input)
    if input.UserInputType == Enum.UserInputType.MouseMovement
        or input.UserInputType == Enum.UserInputType.Touch then

        DragInput = input
    end
end)

UserInputService.InputChanged:Connect(function(input)
    if input == DragInput and Dragging then
        local Delta = input.Position - DragStart
        MainFrame.Position = UDim2.new(
            StartPosition.X.Scale,
            StartPosition.X.Offset + Delta.X,
            StartPosition.Y.Scale,
            StartPosition.Y.Offset + Delta.Y
        )
    end
end)

UserInputService.InputEnded:Connect(function(input)
    if input.UserInputType == Enum.UserInputType.MouseButton1
        or input.UserInputType == Enum.UserInputType.Touch then
        Dragging = false
    end
end)

local function ApplyHover(Button, NormalColor, HoverColor)
    Button.MouseEnter:Connect(function()
        TweenService:Create(
            Button,
            TweenInfo.new(0.2),
            { BackgroundColor3 = HoverColor }
        ):Play()
    end)
    Button.MouseLeave:Connect(function()
        TweenService:Create(
            Button,
            TweenInfo.new(0.2),
            { BackgroundColor3 = NormalColor }
        ):Play()
    end)
end

    --==================================================
    -- SCROLL: mouse wheel + touch
    --==================================================

    local scrollTarget = 0

    local function clampScroll(v)
        local contentHeight = ScriptsScroll.AbsoluteCanvasSize.Y
        local visibleHeight = ScriptsScroll.AbsoluteWindowSize.Y
        local maxScroll = math.max(0, contentHeight - visibleHeight)
        return math.clamp(v, 0, maxScroll)
    end

    local scrollTween

    local function smoothScrollTo(y)
        scrollTarget = clampScroll(y)
        local current = ScriptsScroll.CanvasPosition.Y
        if math.abs(current - scrollTarget) < 0.5 then return end
        if scrollTween then scrollTween:Cancel() end
        scrollTween = TweenService:Create(
            ScriptsScroll,
            TweenInfo.new(0.18, Enum.EasingStyle.Quad, Enum.EasingDirection.Out),
            { CanvasPosition = Vector2.new(0, scrollTarget) }
        )
        scrollTween:Play()
    end

    ScriptsScroll.InputChanged:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseWheel then
            scrollTarget = clampScroll(scrollTarget - input.Position.Z * 40)
            smoothScrollTo(scrollTarget)
        end
    end)

    local touchStartY = nil
    local touchStartCanvas = 0
    local isSwiping = false

    ScriptsScroll.InputBegan:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.Touch then
            touchStartY = input.Position.Y
            touchStartCanvas = ScriptsScroll.CanvasPosition.Y
            isSwiping = true
        end
    end)

    ScriptsScroll.InputChanged:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.Touch and isSwiping then
            local deltaY = touchStartY - input.Position.Y
            ScriptsScroll.CanvasPosition = Vector2.new(0, clampScroll(touchStartCanvas + deltaY))
            scrollTarget = ScriptsScroll.CanvasPosition.Y
        end
    end)

    ScriptsScroll.InputEnded:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.Touch then
            isSwiping = false
        end
    end)
end
