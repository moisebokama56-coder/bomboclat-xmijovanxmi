        math.clamp(textColor.G * 255 * 0.7, 0, 255),
        math.clamp(textColor.B * 255 * 0.7, 0, 255)
    )
    for _, item in pairs(ThemeElements) do
        if item.Type == "Card" then item.Object.BackgroundColor3 = sidebarColor
        elseif item.Type == "Text" then item.Object.TextColor3 = textColor
        elseif item.Type == "Button" then item.Object.BackgroundColor3 = accentColor end
    end
end

        h.TextXAlignment = Enum.TextXAlignment.Left

    KeybindBtn.Parent = Frame; KeybindBtn.BackgroundColor3 = THEME.ToggleOff

    KeybindBtn.TextSize = 11; KeybindBtn.AutoButtonColor = false
    local KbCorner = Instance.new("UICorner"); KbCorner.CornerRadius = UDim.new(0, 6); KbCorner.Parent = KeybindBtn

    local SwitchBg = Instance.new("TextButton")
    SwitchBg.Parent = Frame; SwitchBg.BackgroundColor3 = defaultState and THEME.Accent or THEME.ToggleOff
    SwitchBg.Position = UDim2.new(1, -55, 0, 12); SwitchBg.Size = UDim2.new(0, 44, 0, 24)
    SwitchBg.AutoButtonColor = false; SwitchBg.Text = ""
    local SwitchCorner = Instance.new("UICorner"); SwitchCorner.CornerRadius = UDim.new(1, 0); SwitchCorner.Parent = SwitchBg

    local Indicator = Instance.new("Frame")
    Indicator.Parent = SwitchBg; Indicator.BackgroundColor3 = Color3.fromRGB(255, 255, 255)
    Indicator.Position = defaultState and UDim2.new(1, -21, 0.5, -9) or UDim2.new(0, 3, 0.5, -9)
    Indicator.Size = UDim2.new(0, 18, 0, 18)
    local IndCorner = Instance.new("UICorner"); IndCorner.CornerRadius = UDim.new(1, 0); IndCorner.Parent = Indicator

    local state = defaultState
    local currentKey = nil
    local function trigger()
        state = not state
        TweenService:Create(SwitchBg, TweenInfo.new(0.2), {BackgroundColor3 = state and THEME.Accent or THEME.ToggleOff}):Play()
        TweenService:Create(Indicator, TweenInfo.new(0.2), {Position = state and UDim2.new(1, -21, 0.5, -9) or UDim2.new(0, 3, 0.5, -9)}):Play()
        pcall(callback, state)
    end
    SwitchBg.MouseButton1Click:Connect(trigger)

    local binding = false; local bindConnection = nil
    KeybindBtn.MouseButton1Click:Connect(function()
        if binding then return end
        binding = true; KeybindBtn.Text = "( ... )"
        if bindConnection then bindConnection:Disconnect() end
        bindConnection = UserInputService.InputBegan:Connect(function(input)
            if binding and input.UserInputType == Enum.UserInputType.Keyboard then
                if input.KeyCode == Enum.KeyCode.Escape then currentKey = nil; KeybindBtn.Text = "( ... )"
                else currentKey = input.KeyCode; KeybindBtn.Text = "(" .. input.KeyCode.Name .. ")" end
                binding = false; if bindConnection then bindConnection:Disconnect() end
            end
        end)
    end)

    local kbEntry = {Key = nil, Callback = function() trigger() end}
    table.insert(Keybinds, kbEntry)
    task.spawn(function()
        local lastKey = nil
        while Frame and Frame.Parent do
            if currentKey ~= lastKey then lastKey = currentKey; kbEntry.Key = currentKey end
            task.wait(0.15)
        end
        kbEntry.Key = nil
    end)
    return Frame
end

local function AddSlider(parent, title, hint, min, max, default, callback)
    local Frame = Instance.new("Frame")
    Frame.Parent = parent; Frame.BackgroundColor3 = THEME.Card; Frame.BorderSizePixel = 0; Frame.Size = UDim2.new(1, -10, 0, hint and 72 or 58)
    RegisterElement(Frame, "Card")
    local Corner = Instance.new("UICorner"); Corner.CornerRadius = UDim.new(0, 8); Corner.Parent = Frame
    local Stroke = Instance.new("UIStroke"); Stroke.Parent = Frame; Stroke.Color = THEME.Stroke; Stroke.Transparency = 0.4

    local Label = Instance.new("TextLabel")
    Label.Parent = Frame; Label.BackgroundTransparency = 1
    Label.Position = UDim2.new(0, 15, 0, 6); Label.Size = UDim2.new(1, -30, 0, 20)
    Label.Font = Enum.Font.GothamMedium; Label.Text = title .. ": " .. tostring(default)
    Label.TextColor3 = THEME.Text; Label.TextSize = 13; Label.TextXAlignment = Enum.TextXAlignment.Left
    RegisterElement(Label, "Text")

    local SliderBar = Instance.new("TextButton")
    SliderBar.Parent = Frame; SliderBar.BackgroundColor3 = THEME.ToggleOff; SliderBar.BorderSizePixel = 0
    SliderBar.Position = UDim2.new(0, 15, 0, 36); SliderBar.Size = UDim2.new(1, -30, 0, 8)
    SliderBar.AutoButtonColor = false; SliderBar.Text = ""
    local BarCorner = Instance.new("UICorner"); BarCorner.CornerRadius = UDim.new(1, 0); BarCorner.Parent = SliderBar

    local Fill = Instance.new("Frame")
    Fill.Parent = SliderBar; Fill.BackgroundColor3 = THEME.Accent; Fill.BorderSizePixel = 0
    Fill.Size = UDim2.new((default - min) / (max - min), 0, 1, 0)
    local FillCorner = Instance.new("UICorner"); FillCorner.CornerRadius = UDim.new(1, 0); FillCorner.Parent = Fill

    if hint then
        local h = Instance.new("TextLabel")
        h.Parent = Frame; h.BackgroundTransparency = 1
        h.Position = UDim2.new(0, 15, 0, 50); h.Size = UDim2.new(1, -30, 0, 14)
        h.Font = Enum.Font.Gotham; h.Text = hint; h.TextColor3 = THEME.SubText; h.TextSize = 10
        h.TextXAlignment = Enum.TextXAlignment.Left
    end

    local dragging = false
    local function updateFromX(mouseX)
        local barPos = SliderBar.AbsolutePosition.X
        local barSize = SliderBar.AbsoluteSize.X
        if barSize <= 0 then return end
        local percent = math.clamp((mouseX - barPos) / barSize, 0, 1)
        local value = math.floor(min + (max - min) * percent + 0.5)
        Fill.Size = UDim2.new(percent, 0, 1, 0)
        Label.Text = title .. ": " .. tostring(value)
        pcall(callback, value)
    end
    SliderBar.InputBegan:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
            dragging = true; updateFromX(input.Position.X)
        end
    end)
    SliderBar.InputChanged:Connect(function(input)
        if dragging and (input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch) then
            updateFromX(input.Position.X)
        end
    end)
    UserInputService.InputEnded:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then dragging = false end
    end)
end

local function GetMurdererPlayer()
    for _, p in pairs(Players:GetPlayers()) do
        if p ~= LocalPlayer and p.Character then
            local hasKnife = false
            local bp = p:FindFirstChild("Backpack")
            if bp and bp:FindFirstChild("Knife") then hasKnife = true end
            if p.Character:FindFirstChild("Knife") then hasKnife = true end
            if hasKnife and p.Character:FindFirstChild("HumanoidRootPart") then
                local h = p.Character:FindFirstChildOfClass("Humanoid")
                if h and h.Health > 0 then return p end
            end
        end
    end
    return nil
end

local function GetClosestPlayer()
    local myRoot = LocalPlayer.Character and LocalPlayer.Character:FindFirstChild("HumanoidRootPart")
    if not myRoot then return nil end
    local closest, closestDist = nil, math.huge
    for _, plr in pairs(Players:GetPlayers()) do
        if plr ~= LocalPlayer and plr.Character then
            local h = plr.Character:FindFirstChildOfClass("Humanoid")
            local r = plr.Character:FindFirstChild("HumanoidRootPart")
            if h and r and h.Health > 0 then
                local d = (r.Position - myRoot.Position).Magnitude
                if d < closestDist then closestDist = d; closest = plr end
            end
        end
    end
    return closest
end

local function FindPlayerByName(name)
    if not name or name == "" then return nil end
    name = name:lower()
    for _, plr in pairs(Players:GetPlayers()) do
        if plr ~= LocalPlayer then
            if plr.Name:lower():sub(1, #name) == name or plr.DisplayName:lower():sub(1, #name) == name then
                return plr
            end
        end
    end
    return nil
end

local function GetMurdererDistance()
    local myRoot = LocalPlayer.Character and LocalPlayer.Character:FindFirstChild("HumanoidRootPart")
    local mp = GetMurdererPlayer()
    local mRoot = mp and mp.Character and mp.Character:FindFirstChild("HumanoidRootPart")
    if myRoot and mRoot then return (mRoot.Position - myRoot.Position).Magnitude end
    return nil
end

local function FlingPlayer(targetPlayer)
    if not targetPlayer or not targetPlayer.Character then
        Notify("There was no target to fling.")
        return
    end
    local targetChar = targetPlayer.Character
    local targetHum = targetChar:FindFirstChildOfClass("Humanoid")
    local targetRoot = targetHum and targetHum.RootPart
    local targetHead = targetChar:FindFirstChild("Head")
    local targetAcc = targetChar:FindFirstChildOfClass("Accessory")
    local targetHandle = targetAcc and targetAcc:FindFirstChild("Handle")

    local myChar = LocalPlayer.Character
    local myHum = myChar and myChar:FindFirstChildOfClass("Humanoid")
    local myRoot = myHum and myHum.RootPart
    if not myRoot or not myHum then Notify("Your character is not ready.") return end

    local oldPos = myRoot.CFrame
    local origFPDH = workspace.FallenPartsDestroyHeight

    local function FPos(part, pos, ang)
        myRoot.CFrame = CFrame.new(part.Position) * pos * ang
        myChar:PivotTo(CFrame.new(part.Position) * pos * ang)
        myRoot.Velocity = Vector3.new(9e7, 9e8, 9e7)
        myRoot.RotVelocity = Vector3.new(9e8, 9e8, 9e8)
    end

    local function SFBasePart(basePart)
        local t0 = tick(); local angle = 0
        repeat
            if myRoot and myHum then
                if basePart.Velocity.Magnitude < 50 then
                    angle = angle + 100
                    FPos(basePart, CFrame.new(0, 1.5, 0) + myHum.MoveDirection * basePart.Velocity.Magnitude / 1.25, CFrame.Angles(math.rad(angle), 0, 0)); task.wait()
                    FPos(basePart, CFrame.new(0, -1.5, 0) + myHum.MoveDirection * basePart.Velocity.Magnitude / 1.25, CFrame.Angles(math.rad(angle), 0, 0)); task.wait()
                    FPos(basePart, CFrame.new(2.25, 1.5, -2.25) + myHum.MoveDirection * basePart.Velocity.Magnitude / 1.25, CFrame.Angles(math.rad(angle), 0, 0)); task.wait()
                    FPos(basePart, CFrame.new(-2.25, -1.5, 2.25) + myHum.MoveDirection * basePart.Velocity.Magnitude / 1.25, CFrame.Angles(math.rad(angle), 0, 0)); task.wait()
                else
                    FPos(basePart, CFrame.new(0, 1.5, myHum.WalkSpeed), CFrame.Angles(math.rad(90), 0, 0)); task.wait()
                    FPos(basePart, CFrame.new(0, -1.5, -myHum.WalkSpeed), CFrame.Angles(0, 0, 0)); task.wait()
                    FPos(basePart, CFrame.new(0, 1.5, targetRoot.Velocity.Magnitude / 1.25), CFrame.Angles(math.rad(90), 0, 0)); task.wait()
                    FPos(basePart, CFrame.new(0, -1.5, -targetRoot.Velocity.Magnitude / 1.25), CFrame.Angles(0, 0, 0)); task.wait()
                end
            else break end
        until basePart.Velocity.Magnitude > 500 or basePart.Parent ~= targetChar
           or targetPlayer.Parent ~= Players or targetPlayer.Character ~= targetChar
           or (targetHum and targetHum.Sit) or (myHum and myHum.Health <= 0)
           or tick() > t0 + 2
    end

    workspace.FallenPartsDestroyHeight = 0/0
    local bv = Instance.new("BodyVelocity")
    bv.Name = "NexonFlingVel"; bv.Parent = myRoot
    bv.Velocity = Vector3.new(9e8, 9e8, 9e8)
    bv.MaxForce = Vector3.new(1/0, 1/0, 1/0)
    myHum:SetStateEnabled(Enum.HumanoidStateType.Seated, false)

    if targetRoot and targetHead then
        SFBasePart((targetRoot.CFrame.p - targetHead.CFrame.p).Magnitude > 5 and targetHead or targetRoot)
    elseif targetRoot then SFBasePart(targetRoot)
    elseif targetHead then SFBasePart(targetHead)
    elseif targetHandle then SFBasePart(targetHandle)
    else Notify("I could not find a part to fling.")
    end

    bv:Destroy()
    myHum:SetStateEnabled(Enum.HumanoidStateType.Seated, true)

    repeat
        myRoot.CFrame = oldPos * CFrame.new(0, 0.5, 0)
        myChar:PivotTo(oldPos * CFrame.new(0, 0.5, 0))
        myHum:ChangeState(Enum.HumanoidStateType.GettingUp)
        for _, x in pairs(myChar:GetChildren()) do
            if x:IsA("BasePart") then x.Velocity = Vector3.zero; x.RotVelocity = Vector3.zero end
        end
        task.wait()
    until (myRoot.Position - oldPos.p).Magnitude < 25

    workspace.FallenPartsDestroyHeight = origFPDH
    Notify("Fling complete — thank you for using Nexon Hub.")
end

local FlyUtil = {}
do
    local flying = false
    local maxSpeed = 50
    local accel = 2
    local speed = 0
    local dir = Vector3.new()
    local gyro, vel, hbConn

    local function stopInternal()
        if hbConn then hbConn:Disconnect(); hbConn = nil end
        if gyro then gyro:Destroy(); gyro = nil end
        if vel then vel:Destroy(); vel = nil end
        local char = LocalPlayer.Character
        if char then
            local hum = char:FindFirstChildOfClass("Humanoid")
            if hum then hum.PlatformStand = false end
        end
        flying = false; speed = 0
    end

    local function step()
        local char = LocalPlayer.Character
        if not flying or not char then stopInternal() return end
        local hum = char:FindFirstChildOfClass("Humanoid")
        local root = char:FindFirstChild("HumanoidRootPart")
        local cam = workspace.CurrentCamera
        if not hum or hum.Health <= 0 or not root or not cam then stopInternal() return end

        local moveDir = hum.MoveDirection
        if moveDir.Magnitude > 0.01 then
            speed = math.min(maxSpeed, speed + accel)
            dir = moveDir.Unit
        else
            speed = math.max(0, speed - accel)
        end

        local flat = Vector3.new(dir.X, 0, dir.Z)
        local horizontal = Vector3.zero
        if flat.Magnitude > 0 then horizontal = flat.Unit * speed end

        local camLook = cam.CFrame.LookVector.Unit
        local horizLook = Vector3.new(camLook.X, 0, camLook.Z)
        if horizLook.Magnitude > 0 then horizLook = horizLook.Unit end

        local dot = dir:Dot(camLook)
        local sign = dot < 0 and -1 or 1
        local horizDot = math.abs(dir:Dot(horizLook))
        local vy = camLook.Y * sign * horizDot
        local vertical = vy * speed

        vel.Velocity = Vector3.new(horizontal.X, vertical, horizontal.Z)
        local roll = -math.rad(dot * (speed / maxSpeed) * 30)
        gyro.CFrame = CFrame.new(root.Position, root.Position + camLook) * CFrame.Angles(roll, 0, 0)
    end

    function FlyUtil:Start()
        if flying then return end
        local char = LocalPlayer.Character
        if not char then return end
        local hum = char:FindFirstChildOfClass("Humanoid")
        local root = char:FindFirstChild("HumanoidRootPart")
        if not hum or not root then return end
        flying = true
        gyro = Instance.new("BodyGyro"); gyro.P = 100000
        gyro.MaxTorque = Vector3.new(math.huge, math.huge, math.huge)
        gyro.CFrame = root.CFrame; gyro.Parent = root
        vel = Instance.new("BodyVelocity"); vel.P = 10000
        vel.MaxForce = Vector3.new(math.huge, math.huge, math.huge)
        vel.Velocity = Vector3.zero; vel.Parent = root
        hum.PlatformStand = true
        hbConn = RunService.Heartbeat:Connect(step)
    end
    function FlyUtil:Stop() stopInternal() end
    function FlyUtil:SetMaxSpeed(n) if type(n) == "number" and n >= 0 then maxSpeed = n end end
    function FlyUtil:GetMaxSpeed() return maxSpeed end
    function FlyUtil:IsFlying() return flying end

    LocalPlayer.CharacterRemoving:Connect(function()
        if flying then stopInternal() end
    end)
end

local AntiFling = false
local AntiFlingLastPos = Vector3.zero
local AntiFlingNeut, AntiFlingDetect = nil, nil
local DetectedFlingers = {}

local function StartAntiFling()
    AntiFlingNeut = RunService.Heartbeat:Connect(function()
        local char = LocalPlayer.Character
        local root = char and char.PrimaryPart
        if not root then return end
        if root.AssemblyLinearVelocity.Magnitude > 250 or root.AssemblyAngularVelocity.Magnitude > 250 then
            Notify("You were flung — neutralizing velocity. Please hold on.")
            root.AssemblyLinearVelocity = Vector3.zero
            root.AssemblyAngularVelocity = Vector3.zero
            if AntiFlingLastPos ~= Vector3.zero then root.CFrame = CFrame.new(AntiFlingLastPos) end
        else
            AntiFlingLastPos = root.Position
        end
    end)
    AntiFlingDetect = RunService.Heartbeat:Connect(function()
        for _, pl in ipairs(Players:GetPlayers()) do
            if pl.Character and pl.Character.PrimaryPart and pl.Character:IsDescendantOf(Workspace) then
                local pp = pl.Character.PrimaryPart
                if pp.AssemblyAngularVelocity.Magnitude > 50 or pp.AssemblyLinearVelocity.Magnitude > 100 then
                    if not DetectedFlingers[pl.Name] then
                        DetectedFlingers[pl.Name] = true
                        Notify("A potential flinger was detected: " .. pl.Name)
                    end
                    for _, part in ipairs(pl.Character:GetDescendants()) do
                        if part:IsA("BasePart") then
                            part.CanCollide = false
                            part.AssemblyAngularVelocity = Vector3.zero
                            part.AssemblyLinearVelocity = Vector3.zero
                            pcall(function() part.CustomPhysicalProperties = PhysicalProperties.new(0,0,0) end)
                        end
                    end
                end
            end
        end
    end)
end

local function StopAntiFling()
    if AntiFlingNeut then AntiFlingNeut:Disconnect(); AntiFlingNeut = nil end
    if AntiFlingDetect then AntiFlingDetect:Disconnect(); AntiFlingDetect = nil end
    DetectedFlingers = {}
end

local CoinFarm = false
local CoinWasNoclip = false

local function IsCoin(o)
    if not o:IsA("BasePart") then return false end
    local n = o.Name
    if n == "Coin" or n == "CoinVisual" or n == "CoinPart" or n == "CoinMesh" then return true end
    local p = o.Parent
    if p and (p.Name == "Coin" or p.Name == "CoinContainer" or p.Name == "Coins") then return true end
    return false
end

local function SmoothMove(root, target)
    local start = root.CFrame
    for i = 1, 4 do
        if not CoinFarm then return end
        root.CFrame = start:Lerp(target, i / 4)
        task.wait(0.05)
    end
end

task.spawn(function()
    while task.wait(0.1) do
        if CoinFarm then pcall(function()
            local r = LocalPlayer.Character and LocalPlayer.Character:FindFirstChild("HumanoidRootPart")
            if not r then return end
            local coins = {}
            for _, o in pairs(Workspace:GetDescendants()) do
                if IsCoin(o) then table.insert(coins, o) end
            end
            for _, coin in pairs(coins) do
                if not CoinFarm then break end
                if not coin.Parent then continue end
                local target = CFrame.new(coin.Position + Vector3.new(0, 2, 0))
                if (coin.Position - r.Position).Magnitude > 200 then
                    local mid = r.Position + (coin.Position - r.Position) * 0.5
                    r.CFrame = CFrame.new(mid + Vector3.new(0, 5, 0)); task.wait(0.3)
                    r.CFrame = target; task.wait(0.3)
                else
                    SmoothMove(r, target)
                end
                task.wait(0.15)
            end
        end) end
    end
end)

local KNOWN_MAPS = {
    "House","House2","Beach House","BeachHouse","Mansion","Mansion2","Office",
    "Research","Laboratory","Lab","Cabin","Farm","Ruins","Snow","Snowy","Space",
    "Ship","Yacht","Cruise","Temple","Mineshaft","Mine","Museum","Factory",
    "Asylum","Barn","Volcano","Desert","Villa","Bank","Hotel","School","Prison",
    "Forest","Jungle","Island","Castle"
}

local function IsMapObj(o)
    if not (o:IsA("Folder") or o:IsA("Model")) then return false end
    local n = o.Name:lower()
    local bad = {gameworld=true,lobby=true,players=true,clientparts=true,camera=true,chat=true,terrain=true,effects=true,workspace_lights=true}
    return not bad[n]
end

local function GetRoundMap()
    for _, o in pairs(Workspace:GetChildren()) do
        if IsMapObj(o) then
            local n = o.Name:lower()
            for _, k in pairs(KNOWN_MAPS) do
                if n:find(k:lower(),1,true) then
                    local sp = o:FindFirstChild("Spawn",true) or o:FindFirstChild("SpawnLocation",true)
                    if sp and sp:IsA("BasePart") then return o end
                end
            end
        end
    end
    local best, bestCount = nil, 0
    for _, o in pairs(Workspace:GetChildren()) do
        if IsMapObj(o) then
            local sp = o:FindFirstChild("Spawn",true) or o:FindFirstChild("SpawnLocation",true)
            if sp and sp:IsA("BasePart") then
                local c = 0; for _ in pairs(o:GetDescendants()) do c = c + 1 end
                if c > bestCount then bestCount = c; best = o end
            end
        end
    end
    return best
end

local function TeleportToMap()
    local r = LocalPlayer.Character and LocalPlayer.Character:FindFirstChild("HumanoidRootPart")
    if not r then Notify("Your character is not ready yet.") return end
    local m = GetRoundMap()
    if not m then Notify("I could not detect a round map yet.") return end
    local sp = m:FindFirstChild("Spawn",true) or m:FindFirstChild("SpawnLocation",true)
    if not sp or not sp:IsA("BasePart") then sp = m:FindFirstChildWhichIsA("BasePart",true) end
    if not sp then Notify("That map has no spawn part.") return end
    local was = Noclip; Noclip = true
    r.CFrame = CFrame.new(sp.Position + Vector3.new(0, 5, 0))
    task.wait(0.15)
    if not was then Noclip = false end
    Notify("Teleported to the map: " .. m.Name)
end
